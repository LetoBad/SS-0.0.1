-- ============================================================
-- SANGRESOLIDARIA
-- BASE DE DATOS COMPLETA
-- PostgreSQL / Supabase
-- ============================================================


-- ============================================================
-- 1. LIMPIEZA
-- ============================================================

DROP TABLE IF EXISTS notificaciones CASCADE;
DROP TABLE IF EXISTS donaciones CASCADE;
DROP TABLE IF EXISTS solicitudes CASCADE;
DROP TABLE IF EXISTS donantes CASCADE;
DROP TABLE IF EXISTS usuarios CASCADE;
DROP TABLE IF EXISTS estados_solicitud CASCADE;
DROP TABLE IF EXISTS grupos_sanguineos CASCADE;
DROP TABLE IF EXISTS roles CASCADE;


-- ============================================================
-- 2. ROLES
-- ============================================================

CREATE TABLE roles (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(255)
);


-- ============================================================
-- 3. GRUPOS SANGUÍNEOS
-- ============================================================

CREATE TABLE grupos_sanguineos (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(5) NOT NULL UNIQUE
);


-- ============================================================
-- 4. ESTADOS DE SOLICITUD
-- ============================================================

CREATE TABLE estados_solicitud (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(255)
);


-- ============================================================
-- 5. USUARIOS
-- ============================================================

CREATE TABLE usuarios (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    rol_id BIGINT NOT NULL,

    nombre VARCHAR(100) NOT NULL,

    email VARCHAR(150) NOT NULL UNIQUE,

    telefono VARCHAR(30),

    ciudad VARCHAR(100),

    fecha_nacimiento DATE,

    activo BOOLEAN NOT NULL DEFAULT TRUE,

    creado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_usuario_rol
        FOREIGN KEY (rol_id)
        REFERENCES roles(id)
);


-- ============================================================
-- 6. DONANTES
-- ============================================================

CREATE TABLE donantes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    usuario_id BIGINT NOT NULL UNIQUE,

    grupo_sanguineo_id BIGINT NOT NULL,

    disponible BOOLEAN NOT NULL DEFAULT TRUE,

    ultima_donacion DATE,

    observaciones TEXT,

    creado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_donante_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_donante_grupo
        FOREIGN KEY (grupo_sanguineo_id)
        REFERENCES grupos_sanguineos(id)
);


-- ============================================================
-- 7. SOLICITUDES DE SANGRE
-- ============================================================

CREATE TABLE solicitudes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    usuario_id BIGINT NOT NULL,

    grupo_sanguineo_id BIGINT NOT NULL,

    estado_id BIGINT NOT NULL,

    nombre_paciente VARCHAR(150) NOT NULL,

    cantidad_donantes INTEGER NOT NULL,

    hospital VARCHAR(150) NOT NULL,

    ciudad VARCHAR(100) NOT NULL,

    fecha_necesidad DATE,

    descripcion TEXT,

    creado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_solicitud_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_solicitud_grupo
        FOREIGN KEY (grupo_sanguineo_id)
        REFERENCES grupos_sanguineos(id),

    CONSTRAINT fk_solicitud_estado
        FOREIGN KEY (estado_id)
        REFERENCES estados_solicitud(id),

    CONSTRAINT chk_cantidad_donantes
        CHECK (cantidad_donantes > 0)
);


-- ============================================================
-- 8. DONACIONES
-- ============================================================

CREATE TABLE donaciones (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    donante_id BIGINT NOT NULL,

    solicitud_id BIGINT NOT NULL,

    estado VARCHAR(30) NOT NULL DEFAULT 'registrada',

    fecha_donacion DATE,

    observaciones TEXT,

    creado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_donacion_donante
        FOREIGN KEY (donante_id)
        REFERENCES donantes(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_donacion_solicitud
        FOREIGN KEY (solicitud_id)
        REFERENCES solicitudes(id)
        ON DELETE CASCADE,

    CONSTRAINT chk_estado_donacion
        CHECK (
            estado IN (
                'registrada',
                'confirmada',
                'realizada',
                'cancelada'
            )
        ),

    CONSTRAINT uq_donante_solicitud
        UNIQUE (donante_id, solicitud_id)
);


-- ============================================================
-- 9. NOTIFICACIONES
-- ============================================================

CREATE TABLE notificaciones (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    usuario_id BIGINT NOT NULL,

    titulo VARCHAR(150) NOT NULL,

    mensaje TEXT NOT NULL,

    leida BOOLEAN NOT NULL DEFAULT FALSE,

    tipo VARCHAR(50),

    solicitud_id BIGINT,

    creado_en TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_notificacion_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_notificacion_solicitud
        FOREIGN KEY (solicitud_id)
        REFERENCES solicitudes(id)
        ON DELETE SET NULL
);

-- ============================================================
-- 11. ÍNDICES
-- ============================================================

CREATE INDEX idx_usuarios_email
ON usuarios(email);


CREATE INDEX idx_usuarios_ciudad
ON usuarios(ciudad);


CREATE INDEX idx_donantes_grupo
ON donantes(grupo_sanguineo_id);


CREATE INDEX idx_donantes_disponible
ON donantes(disponible);


CREATE INDEX idx_solicitudes_grupo
ON solicitudes(grupo_sanguineo_id);


CREATE INDEX idx_solicitudes_estado
ON solicitudes(estado_id);


CREATE INDEX idx_solicitudes_ciudad
ON solicitudes(ciudad);


CREATE INDEX idx_solicitudes_fecha
ON solicitudes(fecha_necesidad);


CREATE INDEX idx_donaciones_donante
ON donaciones(donante_id);


CREATE INDEX idx_donaciones_solicitud
ON donaciones(solicitud_id);


CREATE INDEX idx_notificaciones_usuario
ON notificaciones(usuario_id);


-- ============================================================
-- 12. VISTA PARA CONSULTAR SOLICITUDES
-- ============================================================

CREATE VIEW vista_solicitudes AS
SELECT
    s.id,
    s.nombre_paciente,
    gs.nombre AS grupo_sanguineo,
    es.nombre AS estado,
    s.cantidad_donantes,
    s.hospital,
    s.ciudad,
    s.fecha_necesidad,
    s.descripcion,
    s.creado_en,
    u.nombre AS solicitante
FROM solicitudes s
INNER JOIN grupos_sanguineos gs
    ON s.grupo_sanguineo_id = gs.id
INNER JOIN estados_solicitud es
    ON s.estado_id = es.id
INNER JOIN usuarios u
    ON s.usuario_id = u.id;
